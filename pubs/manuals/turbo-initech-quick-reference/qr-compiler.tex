\chapter{The compiler}
\ix{compiler summary}

\QRHead{Compiling}
\QRItem{TPS}{Reads TPSIN.PAS in the current directory; writes TPSOUT.S when
  all four passes succeed.  No options.  Output on COM1.  Exit code 0.}
\QRItem{TPS-LEX-BEGIN ... TPS-LEX-END}{One token per line.}
\QRItem{TPS-PARSE-BEGIN ... TPS-PARSE-OK ... TPS-PARSE-END}{ENTER and EXIT
  of each grammar rule, with the line reached.}
\QRItem{TPS-TYPE-BEGIN ... TPS-TYPE-OK ... TPS-TYPE-END}{SCOPE, SYM and
  FIELD rows: every name, its type, offset and size.}
\QRItem{TPS-GEN-BEGIN\\ TPS-GEN-OK bytes=\textit{n} labels=\textit{n}\\
  TPS-GEN-END}{TPSOUT.S written.}
\QRItem{TPS-\textit{pass}-SKIP \textit{reason}}{A pass skipped after an
  earlier error.}

\QRHead{Capacities}
\noindent{\fontsize{8.5}{10.5}\selectfont
\begin{tabular}{@{}p{25mm}p{10mm}p{29mm}@{}}
SourceMax & 262144 & source characters\\
PoolMax & 6144 & identifier characters\\
NameMax & 704 & different identifiers\\
SymbolMax & 1024 & symbol table rows\\
HeaderParamMax & 32 & parameters, arguments\\
PendingNameMax & 32 & names in one list\\
\end{tabular}\par}

\QRHead{Compiler directives}
\noindent{\fontsize{8.5}{10}\selectfont There are none.  A comment that
begins with \texttt{\{\$} is an ordinary comment.\par}
