\chapter{Creating a Worksheet}

This chapter is for you if you have used a worksheet program before.  In a
few minutes it builds a table of regional sales for a quarter, with a
total for each region.

\begin{Note}
Start Initech 123 with an empty worksheet, as \doctitle{Setting Up Initech
123}, Chapter~2, describes.
\end{Note}

\section{What This Chapter Shows You}
\begin{itemize}
\item Entering labels and numbers
\item Entering a formula
\item Copying a formula
\end{itemize}

\section{Entering the Labels}
\ix{labels}
\begin{Keystrokes}
\Do{Type}{\type{Region}}{in A1, and press \kname{RIGHT}}
\Do{Type}{\type{"Jan}}{press \kname{RIGHT}; \type{"Feb}, \kname{RIGHT};
  \type{"Mar}, \kname{RIGHT}; \type{"Total}, \kname{ENTER}}
\Do{Press}{\fkey{Goto}{F5}}{type \type{a2} and press \kname{ENTER}}
\Do{Type}{\type{North}}{press \kname{DOWN}; then \type{South}, \type{East}
  and \type{West}, each followed by \kname{DOWN} (\kname{ENTER} after the
  last)}
\end{Keystrokes}
The quotation mark aligns a heading at the right of its cell.

\section{Totaling and Copying}
\ix{"@SUM}\ix{/Copy}
\begin{Keystrokes}
\Do{Press}{\fkey{Goto}{F5}}{type \type{e2} and press \kname{ENTER}}
\Do{Type}{\type{@sum(b2..d2)}}{and press \kname{ENTER}}
\Do{Press}{\kname{SLASH}}{type \type{c}, press \kname{ENTER}, type \type{e3}
  and press \kname{ENTER}}
\Do{Press}{\kname{SLASH}}{type \type{c}, type \type{e2..e3} and press
  \kname{ENTER}, type \type{e4} and press \kname{ENTER}}
\end{Keystrokes}

\section{Task Summary}
\begin{LotTable}{p{58mm}p{70mm}}
\LotHead{\textbf{To} & \textbf{You used}}
Enter a label & Type it; a quotation mark first aligns it at the right\\
Enter a number & Type it\\
Go to a cell & GOTO (F5) and the address\\
Add a row of numbers & \texttt{@SUM} and a range\\
Copy a formula & /Copy, the FROM range and the upper left TO cell\LotEnd
\end{LotTable}

\begin{LotTable}{p{58mm}p{70mm}}
