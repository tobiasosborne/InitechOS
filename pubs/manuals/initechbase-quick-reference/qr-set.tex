% Quick Reference: the SET options (Using InitechBase Chapter 5, SET ...).
\QRSection{SET Commands}
\QREntry{ALTERNATE}{set}{SET ALTERNATE TO [<file>] / SET ALTERNATE ON / OFF}{Copies the display to a text file.  OFF.}
\QREntry{CENTURY}{set}{SET CENTURY ON / OFF}{Four-digit years.  OFF.}
\QREntry{CONSOLE}{set}{SET CONSOLE ON / OFF}{Screen output of a program.  ON.}
\QREntry{DATE}{set}{SET DATE AMERICAN / ANSI / BRITISH / FRENCH / GERMAN / ITALIAN / JAPAN / USA}{The form of dates.  AMERICAN.}
\QREntry{DECIMALS}{set}{SET DECIMALS TO [<expN>]}{Decimal places of calculated values, 0 to 15.  2.}
\QREntry{DELETED}{set}{SET DELETED ON / OFF}{Pass over records marked for deletion.  OFF.}
\QREntry{EXACT}{set}{SET EXACT ON / OFF}{Character comparison of whole strings.  OFF.}
\QREntry{FILTER}{set}{SET FILTER TO [<condition>]}{Hides the records that do not meet the condition.}
\QREntry{HEADING}{set}{SET HEADING ON / OFF}{Heading lines of LIST and DISPLAY.  ON.}
\QREntry{INDEX}{set}{SET INDEX TO [<index file list>]}{Opens index files; the first is master.  Without a list, closes them.}
\QREntry{ORDER}{set}{SET ORDER TO [<n>]}{Chooses the master index; 0, record order.}
\QREntry[recorded, not maintained by this version (set.c)]{RELATION}{set}{SET RELATION TO [<key exp> INTO <alias>]}{Links two databases.}
\QREntry{SAFETY}{set}{SET SAFETY ON / OFF}{ZAP asks for confirmation.  ON.}
\QREntry{STATUS}{set}{SET STATUS ON / OFF}{The status bar.  OFF.}
\QREntry[recorded, not acted on by this version (set.c)]{TALK}{set}{SET TALK ON / OFF}{Reporting of results.  ON.}
