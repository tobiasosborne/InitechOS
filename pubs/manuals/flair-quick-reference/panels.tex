\QRHead{Keyboard equivalents}
\noindent A menu shows \CmdKey{N} for \key{Ctrl-N}.  The keys go to the
active application only; a dimmed command's key does nothing.\par
\vspace{2pt}
\begin{tabular}{@{}p{20mm}p{50mm}@{}}
\key{Ctrl-N} & New Folder\\
\key{Ctrl-O} & Open the selected folder\\
\key{Ctrl-W} & Close Window\\
\key{Ctrl-A} & Select All\\
\end{tabular}
\QRHead{In alerts and dialog boxes}
\begin{tabular}{@{}p{20mm}p{50mm}@{}}
\key{Enter} & the default (heavy-bordered) button\\
\key{Esc} & Cancel\\
\key{Up}, \key{Down} & move in the Open and Save lists\\
\end{tabular}
\QRHead{The mouse}
\begin{tabular}{@{}p{20mm}p{50mm}@{}}
Click & select\\
Double-click & open (two clicks within a third of a second)\\
Drag & move an icon or a window; drop on a folder to move into it\\
Shift-click & add to the selection\\
Drag on bare space & select the icons inside the rectangle\\
\end{tabular}
\QRHead{Windows}
\begin{tabular}{@{}p{20mm}p{50mm}@{}}
Title bar & drag to move\\
Close box & left end of the title bar\\
Zoom box & fill the screen; again to restore\\
Collapse box & roll up into the title bar\\
Size box & lower right corner: drag to resize\\
Scroll bars & arrows, grey area, scroll box\\
\end{tabular}
\QRHead{The Trash}
\noindent Drag an icon to the Trash to throw it away.  Double-click the
Trash to see it; drag an item out to keep it.  \textbf{Empty Trash} in the
Special menu deletes it all.\par
\QRHead{Finishing work}
\noindent Click the desktop, then choose \textbf{Shut Down}, \textbf{Restart}
or \textbf{Exit to InitechDOS} from the Special menu.\par
