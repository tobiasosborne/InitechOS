\chapter{The Sample Applications}
\index{sample applications}

Two small applications on the data disk show the way a FLAIR application
behaves: how it opens its window and its menus from the disk, receives the
keys you press and the clicks you make, and quits.  Use them to check that
FLAIR loads applications from the data disk correctly, or to see in their
simplest form the steps every application follows.

\section{C Tenant}
\index{C Tenant}
\Operation{TENANT-REGISTER}
\begin{TaskSteps}
\Step Open the \file{APPS} folder and double-click \file{CTENANT.EXE}.
\Result{A window titled \texttt{C Tenant} opens, with the line \texttt{A C
application, from disk}.}
\Step Press a letter key, such as \key{K}.
\Result{The window shows \texttt{Key k} twice: once in the window's type,
once in reverse in the fixed-width type of the console.}
\Step Click inside the window.
\Result{The window shows \texttt{Click}.}
\end{TaskSteps}

\FlairScreen[scale=4, label=fig:ctenant]{a_ctenant}{C Tenant after a key and
a click}

To quit, choose \menuitem{Quit} from its \menuname{File} menu, press
\key{Ctrl-Q}, or click the close box.

\FlairScreen[scale=4, label=fig:ctenantfile]{a_ctenantfile}{C Tenant's File
menu}
\begin{MenuReference}{CTENANT}{File}
\MenuItemEntry{Quit}{Q}{on}{Quits C Tenant.}
\end{MenuReference}

\section{TenantFix}
\index{TenantFix}
\begin{TaskSteps}
\Step Open the \file{APPS} folder and double-click \file{TENANTFX.EXE}.
\Result{A window titled \texttt{TenantFix} opens, with the lines
\texttt{Loaded from disk} and \texttt{Click here to quit}.}
\end{TaskSteps}

\FlairScreen[label=fig:tfx]{a_tfx}{TenantFix}

\begin{TaskSteps}[resume]
\Step Choose \menuitem{About TenantFix} from the \menuname{Fixture} menu.
\Result{The window shows \texttt{TenantFix 1.0, a disk app}, and a fourth
menu, \menuname{Info}, joins the menu bar.}
\Step Click inside the window.
\Result{TenantFix quits.  Its window closes and the menu bar returns to the
Finder's.}
\end{TaskSteps}
