% A form of the LIST command (compound): its NAME is registered by the
% LIST entry; the inventory keys verbs, as for CLEAR MEMORY (initech-u01i).
\begin{XbaseEntry}{LIST STATUS}{compound}
Lists the open work areas, files, and working environment without pausing.
\Syntax
\SyntaxLine{\kw{LIST STATUS}}
\Usage
The report is the same as for \cmd{DISPLAY STATUS}.  It can be requested
without an open database.  Record pointers, search results, and the selected
work area are preserved.
\SeeAlso{DISPLAY STATUS, SELECT, SET INDEX, SET FILTER, SET RELATION}
\end{XbaseEntry}
