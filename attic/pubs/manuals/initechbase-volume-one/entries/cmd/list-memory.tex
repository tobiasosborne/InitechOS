% Using III+ U5-146/147; inspection mint, byte-exact host and emulator gates.
% A form of the LIST command (compound): its NAME is registered by the
% LIST entry; the inventory keys verbs, as for CLEAR MEMORY (initech-u01i).
\begin{XbaseEntry}{LIST MEMORY}{compound}
Lists the memory variables and their totals without pausing between screens.
\Syntax
\SyntaxLine{\kw{LIST MEMORY}}
\Usage
The entries and totals are the same as for \cmd{DISPLAY MEMORY}, including
hidden outer variables and the program names of PRIVATE variables.
No database need be open.  Variables and work areas are preserved.
\SeeAlso{DISPLAY MEMORY, PUBLIC, PRIVATE, RELEASE}
\end{XbaseEntry}
