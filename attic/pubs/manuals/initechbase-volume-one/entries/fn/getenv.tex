\begin{XbaseEntry}{GETENV}{function}
Returns the value of a DOS environment variable as a character string.
\Syntax
\SyntaxLine{\kw{GETENV(}\xp{expC}\kw{)}}
\Usage
The argument is the environment variable name. If no variable with that name
is set, GETENV() returns an empty string. A call with other than one character
argument raises \xmsg{11}{Invalid function argument.}
\Examples
\DotExample{examples/fn-getenv}
\SeeAlso{RUN, TYPE(), LEN()}
\end{XbaseEntry}
