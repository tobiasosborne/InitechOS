\chapter{The Keys of the Accessories}
\index{keys}

\section{The Calculator}
\begin{center}\small
\begin{tabular}{@{}p{38mm}p{61mm}@{}}
\textbf{Press} & \textbf{To}\\\hline
\key{0} to \key{9}, \key{.} & enter a number\\
\key{+} \key{-} \key{*} \key{/} & choose an operation\\
\key{=} or \key{Enter} & see the result\\
\key{C} or \key{Num Lock} & clear\\
\key{Ctrl-W} & close the Calculator\\
\end{tabular}
\end{center}

\section{InitechText}
\begin{center}\small
\begin{tabular}{@{}p{38mm}p{61mm}@{}}
\textbf{Press} & \textbf{To}\\\hline
\key{Ctrl-N} & start a new document\\
\key{Ctrl-O} & open a document\\
\key{Ctrl-S} & save the document\\
\key{Ctrl-W} or \key{Ctrl-Q} & close the window and quit\\
\key{Ctrl-X}, \key{Ctrl-C}, \key{Ctrl-V} & cut, copy, paste\\
\key{Ctrl-A} & select the whole document\\
arrow keys & move the insertion point\\
\key{Backspace} & erase the character before the insertion point, or the
selection\\
\key{Enter} & end the line\\
\end{tabular}
\end{center}

\section{InitechCalc}
\begin{center}\small
\begin{tabular}{@{}p{38mm}p{61mm}@{}}
\textbf{Press} & \textbf{To}\\\hline
\key{Ctrl-S} & save the tables\\
\key{Ctrl-Q} & quit\\
\key{Left Arrow}, \key{Right Arrow}, \key{Tab} & place the entry and move
across\\
\key{Up Arrow}, \key{Down Arrow} & place the entry and move down or up\\
\key{Backspace} & erase the last character of the entry\\
\key{Esc} & abandon the entry\\
\end{tabular}
\end{center}

\section{InitechPaint and the sample applications}
